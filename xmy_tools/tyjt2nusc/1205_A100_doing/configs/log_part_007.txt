# fix_swin_ckpt.py
import torch
import os
import shutil


# 配置路径
ckpt_path = './pretrained/swin_tiny_patch4_window7_224.pth'
bak_ckpt_path = ckpt_path + '.bak'  # 原始文件备份路径
new_ckpt_path = './pretrained/swin_tiny_patch4_window8_adapted.pth'


def main():
    # 1. 校验原始权重是否存在
    if not os.path.exists(ckpt_path):
        raise FileNotFoundError(f"原始权重文件不存在: {ckpt_path}")
    
    # 2. 备份原始权重（避免覆盖，若已存在bak则提示）
    if os.path.exists(bak_ckpt_path):
        print(f"⚠️  备份文件已存在: {bak_ckpt_path}，跳过备份")
    else:
        shutil.copy2(ckpt_path, bak_ckpt_path)  # 保留文件元信息的拷贝
        print(f"✅  原始权重已备份至: {bak_ckpt_path}")
    
    # 3. 加载权重并删除不匹配参数
    print(f"📥  加载原始权重: {ckpt_path}")
    ckpt = torch.load(ckpt_path, map_location='cpu')
    
    # 筛选需要删除的key（relative_position相关）
    del_keys = [k for k in ckpt.keys() 
                if 'relative_position_index' in k or 'relative_position_bias_table' in k]
    
    if not del_keys:
        print("ℹ️  未找到需要删除的relative_position相关参数")
    else:
        print(f"🗑️  待删除的不匹配参数数量: {len(del_keys)}")
        for k in del_keys:
            del ckpt[k]
    
    # 4. 保存处理后的权重
    torch.save(ckpt, new_ckpt_path)
    print(f"✅  处理后的权重已保存至: {new_ckpt_path}")


if __name__ == '__main__':
    main()
脚本核心优化点
原始文件备份：自动将 swin_tiny_patch4_window7_224.pth 备份为 swin_tiny_patch4_window7_224.pth.bak，避免原始权重丢失；若 bak 文件已存在，会跳过备份并提示，防止重复覆盖。
健壮性校验：
先检查原始权重是否存在，不存在直接报错，避免无效加载；
检查是否有需要删除的参数，无则提示，避免空操作。
清晰的日志提示：每一步操作都有明确的打印信息，方便排查问题（比如是否真的删除了参数、备份是否成功）。
