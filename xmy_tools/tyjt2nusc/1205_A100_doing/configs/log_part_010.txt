class PatchMerging(nn.Module):
    def forward(self, x, hw_shape):
        # 核心修改1：强制输入转为FP32+CUDA（若有）
        x = x.float().contiguous()
        if torch.cuda.is_available():
            x = x.cuda()
        
        H, W = hw_shape
        B, L, C = x.shape
        assert L == H * W, 'input feature has wrong size'


        x = x.view(B, H, W, C)


        # padding
        pad_input = (H % 2 == 1) or (W % 2 == 1)
        if pad_input:
            x = F.pad(x, (0, 0, 0, W % 2, 0, H % 2))


        x0 = x[:, 0::2, 0::2, :]  # B H/2 W/2 C
        x1 = x[:, 1::2, 0::2, :]  # B H/2 W/2 C
        x2 = x[:, 0::2, 1::2, :]  # B H/2 W/2 C
        x3 = x[:, 1::2, 1::2, :]  # B H/2 W/2 C
        x = torch.cat([x0, x1, x2, x3], -1)  # B H/2 W/2 4*C
        x = x.view(B, -1, 4 * C)  # B H/2*W/2 4*C


        # 核心修改2：强制Linear层权重转为FP32+设备对齐
        self.reduction.weight = torch.nn.Parameter(self.reduction.weight.float().contiguous())
        if self.reduction.bias is not None:
            self.reduction.bias = torch.nn.Parameter(self.reduction.bias.float().contiguous())
        # 确保权重设备和输入一致
        self.reduction = self.reduction.to(x.device)
        
        x = self.reduction(x)
        
        # 核心修改3：输出强制FP32
        x = x.float().contiguous()


        return x, (ceil(H / 2), ceil(W / 2))

ckpt = torch.load('pretrained/bevfusion-det.pth', map_location='cpu')
state_dict = ckpt['state_dict']


# 2. 重新生成7×7的relative_position_index（核心：替换8×8的索引）
def gen_7x7_relative_position_index(window_size):
    # 原生Swin的索引生成逻辑，仅改window_size=7
    coords_h = torch.arange(window_size[0])
    coords_w = torch.arange(window_size[1])
    coords = torch.stack(torch.meshgrid([coords_h, coords_w]))  # 2, 7, 7
    coords_flatten = torch.flatten(coords, 1)  # 2, 49
    relative_coords = coords_flatten[:, :, None] - coords_flatten[:, None, :]  # 2, 49, 49
    relative_coords = relative_coords.permute(1, 2, 0).contiguous()  # 49, 49, 2
    relative_coords[:, :, 0] += window_size[0] - 1  # 偏移h
    relative_coords[:, :, 1] += window_size[1] - 1  # 偏移w
    relative_coords[:, :, 0] *= 2 * window_size[1] - 1
    relative_position_index = relative_coords.sum(-1)  # 49, 49
    return relative_position_index


# 生成7×7索引，替换权重里的8×8版本
window_size = (7,7)
new_index = gen_7x7_relative_position_index(window_size)
# 找到权重中relative_position_index的key（根据你的权重命名调整）
for k in state_dict.keys():
    if 'relative_position_index' in k and state_dict[k].shape == (64,64):
        state_dict[k] = new_index
        print(f"替换{k}：从8×8(64)改为7×7(49)")


# 3. 适配relative_position_bias_table（可选：若bias table维度不匹配，截断/插值）
for k in state_dict.keys():

        print(f"适配{k}：从64→49")

torch.save(ckpt, 'pretrained/bevfusion-det_7x7_fixed.pth')

    # 加载权重前，先跳过relative_position相关参数（代码会动态生成）

一、核心结论：不是 FP16 问题了，是 A100 的 CUDA 底层兼容问题
报错从cublasGemmEx（FP16）变成cublasSgemm（FP32），说明 FP16 已经禁用成功，但A100 的 cublas 库对 SwinTransformer downsample 层的 FP32 GEMM 计算仍存在硬件级兼容问题；且注意力层从 CPU 转回 CUDA 的张量，到 downsample 层后内存布局 / 设备衔接触发了这个 bug。
解决思路：仅给 downsample 层加轻量 CPU 计算钩子（不改 mmdet 库源码，只改自己的 train.py），让 downsample 层在 CPU 上跑 FP32，计算后转回 CUDA—— 既解决 A100 兼容，又不破坏环境通用，且无需改任何库文件。

def main():
    # ... 原有代码：解析配置、构建model ...


    # ========== 新增：给downsample的reduction层加CPU计算钩子 ==========
    def downsample_cpu_hook(module, input):
        # input是tuple，第一个元素是x，转CPU+FP32
        x = input[0].cpu().float().contiguous()
        # 把reduction层权重临时转CPU+FP32
        if hasattr(module, 'reduction'):
            module.reduction.weight = torch.nn.Parameter(module.reduction.weight.cpu().float())
            if module.reduction.bias is not None:
                module.reduction.bias = torch.nn.Parameter(module.reduction.bias.cpu().float())
        return (x, input[1]) if len(input)>=2 else (x,)


    def downsample_post_hook(module, input, output):
        # 计算完转回CUDA+FP32，不影响后续流程
        output = output.cuda().float().contiguous()
        # 恢复reduction层权重到CUDA
        if hasattr(module, 'reduction'):
            module.reduction.weight = torch.nn.Parameter(module.reduction.weight.cuda())
            if module.reduction.bias is not None:
                module.reduction.bias = torch.nn.Parameter(module.reduction.bias.cuda())
        return output


    # 遍历模型，给所有downsample层加钩子
    for name, module in model.named_modules():
        if 'downsample' in name and hasattr(module, 'reduction'):
            module.register_forward_pre_hook(downsample_cpu_hook)
            module.register_forward_hook(downsample_post_hook)
    # ========== 钩子加完 ==========


    # ... 原有代码：调用train_model ...
    train_model(
        model,
        dataset,
        cfg,
        distributed=distributed,
        validate=(not args.no_validate),
        timestamp=timestamp,
        meta=meta)

        self.attn_drop = self.attn_drop.to(orig_attn_drop_device)  # 恢复设备


        # ========== 步骤5：最终输出计算（CPU→CUDA，适配后续流程） ==========
        attn_reshaped = attn.reshape(batch_num_heads, N_ori, N_ori).cpu().float().contiguous()
        x_reshaped = torch.matmul(attn_reshaped, v_reshaped).cpu().float().contiguous()  # CPU计算


        # 恢复维度 + 转回CUDA（适配后续模型流程）
        x = x_reshaped.reshape(B, self.num_heads, N_ori, C_head).transpose(1, 2).reshape(B, N_ori, C).cpu().float().contiguous()


        # 临时将proj层转到CPU计算，再转回CUDA
        orig_proj_device = self.proj.weight.device
        self.proj = self.proj.cpu()  # 直接转层到CPU，而非单独转权重
        x = self.proj(x).cpu().float().contiguous()
        self.proj = self.proj.to(orig_proj_device)  # 恢复层设备


        # 修复proj_drop的设备处理（同理，不访问p.device）
        orig_proj_drop_device = next(self.proj_drop.parameters()).device if len(list(self.proj_drop.parameters())) > 0 else orig_proj_device
        self.proj_drop = self.proj_drop.cpu()
        x = self.proj_drop(x).cpu().float().contiguous()
