    cur_coords = None
    for i, arg in enumerate(args):
        if arg.dim() == 3 and arg.shape[1] == 3:  # 匹配3维坐标张量
            arg_cpu = arg.cpu().float()

            cur_coords = cur_coords_cpu.to(orig_device).float()
            args = list(args)
            args[i] = cur_coords
            args = tuple(args)
            break
    # =============================================
    
    # 原有逻辑
    x = super().forward(*args, **kwargs)
    return x

                module.weight = torch.nn.Parameter(module.weight.to(orig_device))
            if hasattr(module, 'bias') and module.bias is not None:
                module.bias = torch.nn.Parameter(module.bias.to(orig_device))

                module.weight = torch.nn.Parameter(module.weight.to(orig_device))
            if hasattr(module, 'bias') and module.bias is not None:
                module.bias = torch.nn.Parameter(module.bias.to(orig_device))
        # ==========================================


        return x


    def forward(self, *args, **kwargs):
        # ========== A100兼容：super().forward转CPU ==========
        orig_device = args[0].device if args else next(self.parameters()).device
        # 输入转CPU
        args_cpu = [arg.cpu().float().contiguous() if isinstance(arg, torch.Tensor) else arg for arg in args]
        kwargs_cpu = {k: v.cpu().float().contiguous() if isinstance(v, torch.Tensor) else v for k, v in kwargs.items()}
        # =====================================================


        x = super().forward(*args_cpu, **kwargs_cpu)  # CPU上执行矩阵乘法


        # ========== 转回CUDA + downsample层兼容 ==========
        x = x.to(orig_device).float().contiguous()
        # downsample层权重临时转CPU计算
        if not isinstance(self.downsample, nn.Identity):
            for module in self.downsample:
                if hasattr(module, 'weight'):
                    module.weight = torch.nn.Parameter(module.weight.cpu().float())
                if hasattr(module, 'bias') and module.bias is not None:
                    module.bias = torch.nn.Parameter(module.bias.cpu().float())
            x_cpu = x.cpu().float().contiguous()
            x = self.downsample(x_cpu).to(orig_device).float().contiguous()
            # 恢复downsample权重
            for module in self.downsample:
                if hasattr(module, 'weight'):
                    module.weight = torch.nn.Parameter(module.weight.to(orig_device))
                if hasattr(module, 'bias') and module.bias is not None:
                    module.bias = torch.nn.Parameter(module.bias.to(orig_device))
        else:
            x = self.downsample(x)
        # ===============================================


        return x

from typing import Tuple


import torch
from mmcv.runner import force_fp32
from torch import nn


from mmdet3d.models.builder import VTRANSFORMS


from .base import BaseDepthTransform


__all__ = ["DepthLSSTransform"]




@VTRANSFORMS.register_module()
class DepthLSSTransform(BaseDepthTransform):
    def __init__(
        self,
        in_channels: int,
        out_channels: int,
        image_size: Tuple[int, int],
        feature_size: Tuple[int, int],
        xbound: Tuple[float, float, float],
        ybound: Tuple[float, float, float],
        zbound: Tuple[float, float, float],
        dbound: Tuple[float, float, float],
        downsample: int = 1,
    ) -> None:
        super().__init__(
            in_channels=in_channels,
            out_channels=out_channels,
            image_size=image_size,
            feature_size=feature_size,
            xbound=xbound,
            ybound=ybound,
            zbound=zbound,
            dbound=dbound,
        )
        self.dtransform = nn.Sequential(
            nn.Conv2d(1, 8, 1),
            nn.BatchNorm2d(8),
            nn.ReLU(True),
            nn.Conv2d(8, 32, 5, stride=4, padding=2),
            nn.BatchNorm2d(32),
            nn.ReLU(True),
            nn.Conv2d(32, 64, 5, stride=2, padding=2),
            nn.BatchNorm2d(64),
            nn.ReLU(True),
        )
        self.depthnet = nn.Sequential(
            nn.Conv2d(in_channels + 64, in_channels, 3, padding=1),
            nn.BatchNorm2d(in_channels),
            nn.ReLU(True),
            nn.Conv2d(in_channels, in_channels, 3, padding=1),
            nn.BatchNorm2d(in_channels),
            nn.ReLU(True),
            nn.Conv2d(in_channels, self.D + self.C, 1),
        )
        if downsample > 1:
            assert downsample == 2, downsample
            self.downsample = nn.Sequential(
                nn.Conv2d(out_channels, out_channels, 3, padding=1, bias=False),
                nn.BatchNorm2d(out_channels),
                nn.ReLU(True),
                nn.Conv2d(
                    out_channels,
                    out_channels,
                    3,
                    stride=downsample,
                    padding=1,
                    bias=False,
                ),
                nn.BatchNorm2d(out_channels),
                nn.ReLU(True),
                nn.Conv2d(out_channels, out_channels, 3, padding=1, bias=False),
                nn.BatchNorm2d(out_channels),
                nn.ReLU(True),
            )
        else:
            self.downsample = nn.Identity()


    @force_fp32()
    def get_cam_feats(self, x, d):
        # ========== A100兼容：所有计算转CPU（核心新增） ==========
        # 1. 保存原始设备，输入转CPU+FP32
        orig_device = x.device
        x = x.cpu().float().contiguous()
        d = d.cpu().float().contiguous()
        
        # 2. 所有层权重临时转CPU
        for module in list(self.dtransform) + list(self.depthnet):
            if hasattr(module, 'weight') and module.weight is not None:
                module.weight = torch.nn.Parameter(module.weight.cpu().float())
            if hasattr(module, 'bias') and module.bias is not None:
                module.bias = torch.nn.Parameter(module.bias.cpu().float())
        # =========================================================


        # 原有业务逻辑（完全不变）
        B, N, C, fH, fW = x.shape
        d = d.view(B * N, *d.shape[2:])
        x = x.view(B * N, C, fH, fW)


        d = self.dtransform(d)
        x = torch.cat([d, x], dim=1)
        x = self.depthnet(x)


        depth = x[:, : self.D].softmax(dim=1)
        x = depth.unsqueeze(1) * x[:, self.D : (self.D + self.C)].unsqueeze(2)


        x = x.view(B, N, self.C, self.D, fH, fW)
        x = x.permute(0, 1, 3, 4, 5, 2)


        # ========== 转回CUDA，恢复权重设备 ==========
        x = x.to(orig_device).float().contiguous()
        for module in list(self.dtransform) + list(self.depthnet):
            if hasattr(module, 'weight') and module.weight is not None:
                module.weight = torch.nn.Parameter(module.weight.to(orig_device))
            if hasattr(module, 'bias') and module.bias is not None:
                module.bias = torch.nn.Parameter(module.bias.to(orig_device))
        # ==========================================


        return x


    def forward(self, *args, **kwargs):
        # ========== A100兼容：super().forward转CPU ==========
        orig_device = args[0].device if args else next(self.parameters()).device
        # 输入张量转CPU
        args_cpu = [a.cpu().float().contiguous() if isinstance(a, torch.Tensor) else a for a in args]
        kwargs_cpu = {k: v.cpu().float().contiguous() if isinstance(v, torch.Tensor) else v for k, v in kwargs.items()}
        # =====================================================


        # 父类forward在CPU执行（规避矩阵乘法报错）
        x = super().forward(*args_cpu, **kwargs_cpu)
        
        # ========== 转回CUDA + downsample层兼容 ==========
        x = x.to(orig_device).float().contiguous()
        # downsample层临时转CPU计算
        if not isinstance(self.downsample, nn.Identity):
            # downsample权重转CPU
            for module in self.downsample:
                if hasattr(module, 'weight'):
                    module.weight = torch.nn.Parameter(module.weight.cpu().float())
                if hasattr(module, 'bias') and module.bias is not None:
                    module.bias = torch.nn.Parameter(module.bias.cpu().float())
            # CPU执行downsample
            x_cpu = x.cpu().float().contiguous()
            x = self.downsample(x_cpu).to(orig_device).float().contiguous()
            # 恢复downsample权重到CUDA
            for module in self.downsample:
                if hasattr(module, 'weight'):
                    module.weight = torch.nn.Parameter(module.weight.to(orig_device))
                if hasattr(module, 'bias') and module.bias is not None:
                    module.bias = torch.nn.Parameter(module.bias.to(orig_device))
        else:
            x = self.downsample(x)
        # ===============================================


        return x
