@force_fp32()
def get_cam_feats(self, img, depth, mats_dict): 
    # 参数名映射
    x = img
    d = depth
    orig_device = x.device
    
    # 输入转CPU
    x = x.cpu().float().contiguous()
    d = d.cpu().float().contiguous()
    
    # ========== 新增：保存所有参数+Buffer的原设备 ==========
    # 处理Module的Parameter（weight/bias）和Buffer（running_mean/running_var）
    def save_module_devices(module):
        dev_map = []
        # 保存Parameter
        for name, param in module.named_parameters(recurse=False):
            if param is not None:
                dev_map.append(('param', name, param.device))
        # 保存Buffer（关键！处理BatchNorm的running_mean/running_var）
        for name, buf in module.named_buffers(recurse=False):
            if buf is not None:
                dev_map.append(('buffer', name, buf.device))
        return dev_map


    def move_module_to_device(module, dev_map, target_dev):
        for typ, name, _ in dev_map:
            if typ == 'param':
                param = getattr(module, name)
                setattr(module, name, torch.nn.Parameter(param.to(target_dev).float()))
            elif typ == 'buffer':
                buf = getattr(module, name)
                setattr(module, name, buf.to(target_dev).float())


    # 收集dtransform和depthnet的设备信息
    dev_maps = []
    for module in list(self.dtransform) + list(self.depthnet):
        dev_maps.append((module, save_module_devices(module)))
        # 转到CPU
        move_module_to_device(module, dev_maps[-1][1], torch.device('cpu'))
    # =========================================================


    # 原有业务逻辑
    B, N, C, fH, fW = x.shape
    d = d.view(B * N, *d.shape[2:])
    x = x.view(B * N, C, fH, fW)


    d = self.dtransform(d)  # 现在BatchNorm的Buffer在CPU，输入d也在CPU，无冲突
    x = torch.cat([d, x], dim=1)
    x = self.depthnet(x)


    depth_out = x[:, : self.D].softmax(dim=1)
    x = depth_out.unsqueeze(1) * x[:, self.D : (self.D + self.C)].unsqueeze(2)


    x = x.view(B, N, self.C, self.D, fH, fW)
    x = x.permute(0, 1, 3, 4, 5, 2)


    # ========== 还原所有Parameter+Buffer到原设备 ==========
    for module, dev_map in dev_maps:
        move_module_to_device(module, dev_map, orig_device)
    # =========================================================


    # 结果转回CUDA
    x = x.to(orig_device).float().contiguous()
    return x


def forward(self, *args, **kwargs):
    orig_device = args[0].device if args else next(self.parameters()).device
    # 输入转CPU
    args_cpu = [a.cpu().float().contiguous() if isinstance(a, torch.Tensor) else a for a in args]
    kwargs_cpu = {k: v.cpu().float().contiguous() if isinstance(v, torch.Tensor) else v for k, v in kwargs.items()}


    # 父类forward在CPU执行
    x = super().forward(*args_cpu, **kwargs_cpu)
    
    # 转回CUDA
    x = x.to(orig_device).float().contiguous()


    # ========== 新增：downsample层处理Buffer ==========
    if not isinstance(self.downsample, nn.Identity):
        # 保存downsample的Parameter+Buffer设备
        down_dev_maps = []
        for module in self.downsample:
            down_dev_maps.append((module, save_module_devices(module)))
            move_module_to_device(module, down_dev_maps[-1][1], torch.device('cpu'))
        
        # CPU执行downsample
        x_cpu = x.cpu().float().contiguous()
        x = self.downsample(x_cpu).to(orig_device).float().contiguous()
        
        # 还原downsample的Parameter+Buffer
        for module, dev_map in down_dev_maps:
            move_module_to_device(module, dev_map, orig_device)
    else:
        x = self.downsample(x)
    # =========================================================


    return x

def bev_pool(self, geom, x):

    dev = geom.device  # 获取geom_feats的设备（CPU）
    # 将所有静态张量转到当前设备
    self.bx = self.bx.to(dev)
    self.dx = self.dx.to(dev)
    self.by = self.by.to(dev)
    self.dy = self.dy.to(dev)
    self.bz = self.bz.to(dev)
    self.dz = self.dz.to(dev)
    # ================================================================


    B, N, D, H, W, C = x.shape
    x = x.view(B*N, D, H, W, C)


    geom_feats = geom.view(B*N, D, H, W, 3)
    # 修复后：geom_feats和bx/dx设备一致，无冲突
    geom_feats = ((geom_feats - (self.bx - self.dx / 2.0)) / self.dx).long()
    geom_feats[..., 1] = ((geom_feats[..., 1] - (self.by - self.dy / 2.0)) / self.dy).long()
    geom_feats[..., 2] = ((geom_feats[..., 2] - (self.bz - self.dz / 2.0)) / self.dz).long()


    # 边界裁剪（原有逻辑不变）
    mask = ((geom_feats[..., 0] >= 0) 
            & (geom_feats[..., 0] < self.nx) 
            & (geom_feats[..., 1] >= 0) 
            & (geom_feats[..., 1] < self.ny) 
            & (geom_feats[..., 2] >= 0) 
            & (geom_feats[..., 2] < self.nz))
    mask = mask.view(B*N, D, H, W, 1)
    x = x * mask


    # 计算bev坐标（原有逻辑不变）
    coord = geom_feats[..., 0] + self.nx * geom_feats[..., 1] + self.nx * self.ny * geom_feats[..., 2]
    coord = coord.view(B*N, D*H*W)
    x = x.view(B*N, D*H*W, C)


    # BEV池化（原有逻辑不变）
    bev = torch.zeros((B*N, self.nx*self.ny*self.nz, C), device=x.device, dtype=x.dtype)
    bev.scatter_add_(1, coord.unsqueeze(-1).repeat(1, 1, C), x)
    bev = bev.view(B, N, self.nz, self.ny, self.nx, C)
    bev = bev.permute(0, 1, 5, 2, 3, 4).contiguous()


    return bev

@force_fp32()
def get_cam_feats(self, img, depth, mats_dict): 
    x = img
    d = depth
    B, N, C, fH, fW = x.shape
    d = d.view(B * N, *d.shape[2:])
    x = x.view(B * N, C, fH, fW)

    d = self.dtransform(d)
    x = torch.cat([d, x], dim=1)
    x = self.depthnet(x)

    depth_out = x[:, : self.D].softmax(dim=1)
    x = depth_out.unsqueeze(1) * x[:, self.D : (self.D + self.C)].unsqueeze(2)

    x = x.view(B, N, self.C, self.D, fH, fW)
    x = x.permute(0, 1, 3, 4, 5, 2)
    return x

def forward(self, *args, **kwargs):
    x = super().forward(*args, **kwargs)
    x = self.downsample(x)
    return x

是谁和谁的版本有兼容问题？

第一重核心冲突（直接导致 cublas 报错）
PyTorch 1.12.1 ↔ A100 显卡 + CUDA 11.6（PyTorch 内置）
冲突双方：PyTorch 1.12.1（框架） ↔ A100 显卡 + PyTorch 内置的 CUDA 11.6 库；
冲突表现：A100 执行 FP16 矩阵运算时，PyTorch 1.12.1 无法正确调用 CUDA 11.6 的 cuBLAS LT 接口，直接触发cublasSgemm failed报错；

第二重隐性冲突（加剧设备不匹配报错）
PyTorch 内置 CUDA 11.6 ↔ 系统 CUDA 12.8
冲突双方：PyTorch 内部依赖的 CUDA 11.6 库 ↔ 服务器系统安装的 CUDA 12.8；
冲突表现：看似张量在cuda:0，但 A100 的 GPU 内核调用会因 CUDA 版本割裂，出现 “张量设备显示 CUDA 但实际绑定 CPU”，触发Expected all tensors to be on the same device报错；
官方结论：PyTorch 不依赖系统 CUDA，但系统 CUDA 与内置 CUDA 版本差≥1 个大版本（11.6→12.8），会导致 A100 的设备逻辑隐性冲突。
唯一无冲突的组合（对照参考）

问题 (2) 先讲清楚：“CUDA 内置：11.8” 是什么？（核心：和系统 nvcc 无关，不用重装！）
你完全不用重装系统的 nvcc -V 11.6，也不用管网络差的问题 ——PyTorch 的 “CUDA 内置” 和系统的 CUDA Toolkit（nvcc）是两回事，拆解如下：
概念	含义	是否需要动系统？	对你的影响
CUDA内置:11.8（PyTorch）	PyTorch 安装包自带的 “CUDA 运行时库”（仅给 PyTorch 自己用，比如矩阵运算、GPU 调用）	❌ 完全不用	升级 PyTorch 后，仅改这个内置库，不碰系统任何东西
nvcc -V 11.6（系统）	系统级的 CUDA Toolkit（包含编译器 nvcc、开发库等，给 C++/CUDA 编程用）	❌ 不用重装 / 修改	你的系统 nvcc 保持 11.6 不变，完全不影响 PyTorch 使用
关键结论：PyTorch 运行 GPU 代码时，只依赖自身内置的 CUDA 运行时库，不依赖系统的 nvcc 版本（哪怕系统 nvcc 是 11.6，PyTorch 内置 11.8 也能正常跑）。你不用动系统的 CUDA/nvcc，仅升级 PyTorch 即可，网络差也没关系（PyTorch 安装包只有几百 MB，远小于系统 CUDA 的几个 GB）。

2.  与混合精度训练的结合：在深度学习领域，FP16的真正优势通常体现在混合精度训练中。这种方法并非将整个模型都转换为FP16，而是智能地结合FP16和FP32：用FP16进行大部分计算以获得速度提升，同时用FP32来维护梯度和关键权重的精度，从而在保证模型收敛性和准确性的前提下，实现显著的加速。

公开资料中提到的“自动混合精度”正是为了简化这一过程，让研究人员只需添加少量代码，就能利用FP16将性能提升2倍，同时保持模型的准确性。
