class BaseDepthTransform(BaseTransform):
    @force_fp32()
    def forward(
        self,
        img,
        points,
        radar, 
        sensor2ego,
        lidar2ego,
        lidar2camera,
        lidar2image,
        cam_intrinsic,
        camera2lidar,
        img_aug_matrix,
        lidar_aug_matrix,
        metas,
        **kwargs,
    ):
        rots = sensor2ego[..., :3, :3]
        trans = sensor2ego[..., :3, 3]
        intrins = cam_intrinsic[..., :3, :3]
        post_rots = img_aug_matrix[..., :3, :3]
        post_trans = img_aug_matrix[..., :3, 3]
        lidar2ego_rots = lidar2ego[..., :3, :3]
        lidar2ego_trans = lidar2ego[..., :3, 3]
        camera2lidar_rots = camera2lidar[..., :3, :3]
        camera2lidar_trans = camera2lidar[..., :3, 3]


        if self.use_points == 'radar':
            points = radar


        if self.height_expand:
            for b in range(len(points)):
                points_repeated = points[b].repeat_interleave(8, dim=0)
                points_repeated[:, 2] = torch.arange(0.25, 2.25, 0.25).repeat(points[b].shape[0])
                points[b] = points_repeated


        batch_size = len(points)
        depth_in_channels = 1 if self.depth_input=='scalar' else self.D
        if self.add_depth_features:
            depth_in_channels += points[0].shape[1]


        depth = torch.zeros(batch_size, img.shape[1], depth_in_channels, *self.image_size, device=points[0].device)




        for b in range(batch_size):
            cur_coords = points[b][:, :3]
            cur_img_aug_matrix = img_aug_matrix[b]
            cur_lidar_aug_matrix = lidar_aug_matrix[b]
            cur_lidar2image = lidar2image[b]


            # ========== 新增：A100兼容 - 矩阵运算转CPU（核心修改） ==========
            orig_device = cur_lidar_aug_matrix.device  # 保存原始CUDA设备
            # 1. 所有参与矩阵运算的张量转CPU+FP32
            cur_coords_cpu = cur_coords.cpu().float()
            cur_lidar_aug_matrix_cpu = cur_lidar_aug_matrix.cpu().float()
            cur_lidar2image_cpu = cur_lidar2image.cpu().float()
            cur_img_aug_matrix_cpu = cur_img_aug_matrix.cpu().float()
            # =============================================================


            # inverse aug（CPU执行，规避cublas报错）
            cur_coords_cpu -= cur_lidar_aug_matrix_cpu[:3, 3]
            cur_coords_cpu = torch.inverse(cur_lidar_aug_matrix_cpu[:3, :3]).matmul(
                cur_coords_cpu.transpose(1, 0)
            )
            # lidar2image（CPU执行）
            cur_coords_cpu = cur_lidar2image_cpu[:, :3, :3].matmul(cur_coords_cpu)
            cur_coords_cpu += cur_lidar2image_cpu[:, :3, 3].reshape(-1, 3, 1)
            # get 2d coords（CPU执行）
            dist_cpu = cur_coords_cpu[:, 2, :]
            cur_coords_cpu[:, 2, :] = torch.clamp(cur_coords_cpu[:, 2, :], 1e-5, 1e5)
            cur_coords_cpu[:, :2, :] /= cur_coords_cpu[:, 2:3, :]


            # imgaug（CPU执行）
            cur_coords_cpu = cur_img_aug_matrix_cpu[:, :3, :3].matmul(cur_coords_cpu)
            cur_coords_cpu += cur_img_aug_matrix_cpu[:, :3, 3].reshape(-1, 3, 1)
            cur_coords_cpu = cur_coords_cpu[:, :2, :].transpose(1, 2)


            # ========== 新增：转回CUDA，恢复原始设备 ==========
            cur_coords = cur_coords_cpu.to(orig_device).float()
            dist = dist_cpu.to(orig_device).float()
            # =================================================


            # normalize coords for grid sample
            cur_coords = cur_coords[..., [1, 0]]


            on_img = (
                (cur_coords[..., 0] < self.image_size[0])
                & (cur_coords[..., 0] >= 0)
                & (cur_coords[..., 1] < self.image_size[1])
                & (cur_coords[..., 1] >= 0)
            )
            for c in range(on_img.shape[0]):
                masked_coords = cur_coords[c, on_img[c]].long()
                masked_dist = dist[c, on_img[c]]


                if self.depth_input == 'scalar':
                    depth[b, c, 0, masked_coords[:, 0], masked_coords[:, 1]] = masked_dist
                elif self.depth_input == 'one-hot':
                    # Clamp depths that are too big to D
                    # These can arise when the point range filter is different from the dbound. 
                    masked_dist = torch.clamp(masked_dist, max=self.D-1)
                    depth[b, c, masked_dist.long(), masked_coords[:, 0], masked_coords[:, 1]] = 1.0


                if self.add_depth_features:
                    # 修复boolmask2idx未定义问题（若有），保持原有逻辑
                    depth[b, c, -points[b].shape[-1]:, masked_coords[:, 0], masked_coords[:, 1]] = points[b][on_img[c]].transpose(0,1)


        extra_rots = lidar_aug_matrix[..., :3, :3]
        extra_trans = lidar_aug_matrix[..., :3, 3]
        geom = self.get_geometry(
            camera2lidar_rots,
            camera2lidar_trans,
            intrins,
            post_rots,
            post_trans,
            extra_rots=extra_rots,
            extra_trans=extra_trans,
        )

            'intrin_mats': intrins, 
            'ida_mats': img_aug_matrix, 
            'bda_mat': lidar_aug_matrix,
            'sensor2ego_mats': sensor2ego, 
        }
        x = self.get_cam_feats(img, depth, mats_dict)


        use_depth = False
        if type(x) == tuple:
            x, depth = x 
            use_depth = True
        
        x = self.bev_pool(geom, x)

            return x, depth

核心报错原因：height_expand 中 torch.arange 创建的张量是 CPU 设备，导致 points[b] 被污染为 CPU 张量，进而引发后续赋值时的设备不匹配！
你之前的 CPU/GPU 切换逻辑没问题，但 height_expand 里的 torch.arange 未指定设备，默认创建在 CPU，赋值给 points_repeated（原本是 CUDA）后，points[b] 变成 CPU 张量，最终：
depth 张量基于 points[0].device 创建（变成 CPU）；
而 masked_dist 是转回 CUDA 的张量；
赋值时 depth (CPU) ← masked_dist (CUDA)，触发设备不匹配报错。
🚀 终极修复代码（修改 BaseDepthTransform 类）
只需在 height_expand 部分给 torch.arange 指定设备，同时统一所有张量设备，修复后代码如下：

class BaseDepthTransform(BaseTransform):
    @force_fp32()
    def forward(
        self,
        img,
        points,
        radar, 
        sensor2ego,
        lidar2ego,
        lidar2camera,
        lidar2image,
        cam_intrinsic,
        camera2lidar,
        img_aug_matrix,
        lidar_aug_matrix,
        metas,
        **kwargs,
    ):
        rots = sensor2ego[..., :3, :3]
        trans = sensor2ego[..., :3, 3]
        intrins = cam_intrinsic[..., :3, :3]
        post_rots = img_aug_matrix[..., :3, :3]
        post_trans = img_aug_matrix[..., :3, 3]
        lidar2ego_rots = lidar2ego[..., :3, :3]
        lidar2ego_trans = lidar2ego[..., :3, 3]
        camera2lidar_rots = camera2lidar[..., :3, :3]
        camera2lidar_trans = camera2lidar[..., :3, 3]


        if self.use_points == 'radar':
            points = radar


        # ========== 修复1：height_expand 中指定张量设备（核心！） ==========
        if self.height_expand:
            for b in range(len(points)):
                # 获取points[b]的设备，确保arange创建在同一设备
                device = points[b].device
                points_repeated = points[b].repeat_interleave(8, dim=0)
                # 关键：指定arange的device为points[b].device，避免污染为CPU
                z_vals = torch.arange(0.25, 2.25, 0.25, device=device).repeat(points[b].shape[0])
                points_repeated[:, 2] = z_vals
                points[b] = points_repeated
        # ================================================================


        batch_size = len(points)
        depth_in_channels = 1 if self.depth_input=='scalar' else self.D
        if self.add_depth_features:
            depth_in_channels += points[0].shape[1]


        # ========== 修复2：显式指定depth的device为CUDA（兜底） ==========
        depth_device = points[0].device if points else img.device
        depth = torch.zeros(batch_size, img.shape[1], depth_in_channels, *self.image_size, device=depth_device)
        # ================================================================


        for b in range(batch_size):
            cur_coords = points[b][:, :3]
            cur_img_aug_matrix = img_aug_matrix[b]
            cur_lidar_aug_matrix = lidar_aug_matrix[b]
            cur_lidar2image = lidar2image[b]


            # ========== A100兼容 - 矩阵运算转CPU ==========
            orig_device = cur_lidar_aug_matrix.device
            cur_coords_cpu = cur_coords.cpu().float()
            cur_lidar_aug_matrix_cpu = cur_lidar_aug_matrix.cpu().float()
            cur_lidar2image_cpu = cur_lidar2image.cpu().float()
            cur_img_aug_matrix_cpu = cur_img_aug_matrix.cpu().float()
            # =================================================


            # inverse aug（CPU执行）
            cur_coords_cpu -= cur_lidar_aug_matrix_cpu[:3, 3]
            cur_coords_cpu = torch.inverse(cur_lidar_aug_matrix_cpu[:3, :3]).matmul(
                cur_coords_cpu.transpose(1, 0)
            )
            # lidar2image（CPU执行）
            cur_coords_cpu = cur_lidar2image_cpu[:, :3, :3].matmul(cur_coords_cpu)
            cur_coords_cpu += cur_lidar2image_cpu[:, :3, 3].reshape(-1, 3, 1)
            # get 2d coords（CPU执行）
            dist_cpu = cur_coords_cpu[:, 2, :]
            cur_coords_cpu[:, 2, :] = torch.clamp(cur_coords_cpu[:, 2, :], 1e-5, 1e5)
            cur_coords_cpu[:, :2, :] /= cur_coords_cpu[:, 2:3, :]


            # imgaug（CPU执行）
            cur_coords_cpu = cur_img_aug_matrix_cpu[:, :3, :3].matmul(cur_coords_cpu)
            cur_coords_cpu += cur_img_aug_matrix_cpu[:, :3, 3].reshape(-1, 3, 1)
            cur_coords_cpu = cur_coords_cpu[:, :2, :].transpose(1, 2)


            # ========== 转回CUDA，确保设备统一 ==========
            cur_coords = cur_coords_cpu.to(orig_device).float()
            dist = dist_cpu.to(orig_device).float()
            # =================================================


            # normalize coords for grid sample
            cur_coords = cur_coords[..., [1, 0]]


            on_img = (
                (cur_coords[..., 0] < self.image_size[0])
                & (cur_coords[..., 0] >= 0)
                & (cur_coords[..., 1] < self.image_size[1])
                & (cur_coords[..., 1] >= 0)
            )
            # ========== 修复3：确保on_img和索引张量设备统一 ==========
            on_img = on_img.to(depth_device)
            # =================================================


            for c in range(on_img.shape[0]):
                masked_coords = cur_coords[c, on_img[c]].long().to(depth_device)
                masked_dist = dist[c, on_img[c]].to(depth_device)


                if self.depth_input == 'scalar':
                    # 现在depth和masked_dist都在同一设备（CUDA），赋值无冲突
                    depth[b, c, 0, masked_coords[:, 0], masked_coords[:, 1]] = masked_dist
                elif self.depth_input == 'one-hot':
                    masked_dist = torch.clamp(masked_dist, max=self.D-1).to(depth_device)
                    depth[b, c, masked_dist.long(), masked_coords[:, 0], masked_coords[:, 1]] = 1.0


                if self.add_depth_features:
                    # 确保points[b]的索引张量设备统一
                    idx = on_img[c].to(points[b].device)
                    depth[b, c, -points[b].shape[-1]:, masked_coords[:, 0], masked_coords[:, 1]] = points[b][idx].transpose(0,1).to(depth_device)


        extra_rots = lidar_aug_matrix[..., :3, :3]
        extra_trans = lidar_aug_matrix[..., :3, 3]
        geom = self.get_geometry(
            camera2lidar_rots,
            camera2lidar_trans,
            intrins,
            post_rots,
            post_trans,
            extra_rots=extra_rots,
            extra_trans=extra_trans,
        )


        mats_dict = {
            'intrin_mats': intrins, 
            'ida_mats': img_aug_matrix, 
            'bda_mat': lidar_aug_matrix,
            'sensor2ego_mats': sensor2ego, 
        }
        x = self.get_cam_feats(img, depth, mats_dict)


        use_depth = False
        if type(x) == tuple:
            x, depth = x 
            use_depth = True
        
        x = self.bev_pool(geom, x)


        if use_depth:
            return x, depth 
        else:
            return x
✅ 修复关键点说明
核心修复（height_expand）：
torch.arange(0.25, 2.25, 0.25, device=device) → 指定设备为 points[b].device，避免 points[b] 被 CPU 张量污染；

显式指定 depth 的设备为 points[0].device 或 img.device，确保和后续赋值的张量设备一致；
索引张量设备统一：
on_img/masked_coords/masked_dist 都显式转到 depth_device，彻底避免设备不匹配。

核心报错原因：self.frustum 张量默认创建在 CPU 设备，而 post_trans 是 CUDA 张量，两者运算时设备不匹配！
self.frustum 是在 create_frustum 方法中创建的 nn.Parameter，默认在 CPU；而 get_geometry 中的 post_trans/camera2lidar_rots 等张量是 CUDA 设备，减法运算 points = self.frustum - post_trans.view(...) 触发设备不匹配报错。
🚀 终极修复代码（修改 BaseTransform 类的 get_geometry 方法）

    # ... 其余代码不变，仅修改 get_geometry 方法 ...


    @force_fp32()
    def get_geometry(
        self,
        camera2lidar_rots,
        camera2lidar_trans,
        intrins,
        post_rots,
        post_trans,
        **kwargs,
    ):
        B, N, _ = camera2lidar_trans.shape


        # ========== 核心修复：将 frustum 转到 post_trans 的设备（CUDA） ==========
        device = post_trans.device  # 获取CUDA设备
        frustum = self.frustum.to(device)  # 把CPU的frustum转到CUDA
        # ================================================================


        # undo post-transformation（使用转设备后的frustum）
        # B x N x D x H x W x 3
        points = frustum - post_trans.view(B, N, 1, 1, 1, 3)  # 不再有设备冲突
        points = (
            torch.inverse(post_rots)
            .view(B, N, 1, 1, 1, 3, 3)
            .matmul(points.unsqueeze(-1))
        )
        # cam_to_lidar
        points = torch.cat(
            (
                points[:, :, :, :, :, :2] * points[:, :, :, :, :, 2:3],
                points[:, :, :, :, :, 2:3],
            ),
            5,
        )
        combine = camera2lidar_rots.matmul(torch.inverse(intrins))
        points = combine.view(B, N, 1, 1, 1, 3, 3).matmul(points).squeeze(-1)
        points += camera2lidar_trans.view(B, N, 1, 1, 1, 3)


        if "extra_rots" in kwargs:
            extra_rots = kwargs["extra_rots"]
            points = (
                extra_rots.view(B, 1, 1, 1, 1, 3, 3)
                .repeat(1, N, 1, 1, 1, 1, 1)
                .matmul(points.unsqueeze(-1))
                .squeeze(-1)
            )
        if "extra_trans" in kwargs:
            extra_trans = kwargs["extra_trans"]
            points += extra_trans.view(B, 1, 1, 1, 1, 3).repeat(1, N, 1, 1, 1, 1)


        return points
