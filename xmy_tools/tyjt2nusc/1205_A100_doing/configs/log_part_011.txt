class PatchMerging(nn.Module):
    def forward(self, x, hw_shape):
        # ========== 硬编码：强制转CPU+FP32（和注意力层逻辑对齐） ==========
        # 1. 保存原始设备（CUDA），输入转CPU+FP32
        orig_device = x.device
        x = x.cpu().float().contiguous()
        
        # 2. reduction层权重转CPU+FP32
        self.reduction.weight = torch.nn.Parameter(self.reduction.weight.cpu().float())
        if self.reduction.bias is not None:
            self.reduction.bias = torch.nn.Parameter(self.reduction.bias.cpu().float())
        
        # ========== 原有逻辑保留（仅设备为CPU） ==========
        H, W = hw_shape
        B, L, C = x.shape
        assert L == H * W, 'input feature has wrong size'


        x = x.view(B, H, W, C)


        # padding
        pad_input = (H % 2 == 1) or (W % 2 == 1)
        if pad_input:
            x = F.pad(x, (0, 0, 0, W % 2, 0, H % 2))


        x0 = x[:, 0::2, 0::2, :]  # B H/2 W/2 C
        x1 = x[:, 1::2, 0::2, :]  # B H/2 W/2 C
        x2 = x[:, 0::2, 1::2, :]  # B H/2 W/2 C
        x3 = x[:, 1::2, 1::2, :]  # B H/2 W/2 C
        x = torch.cat([x0, x1, x2, x3], -1)  # B H/2 W/2 4*C
        x = x.view(B, -1, 4 * C)  # B H/2*W/2 4*C


        # CPU上执行reduction计算
        x = self.reduction(x).float().contiguous()
        
        # ========== 转回CUDA，恢复权重设备 ==========
        x = x.to(orig_device).float().contiguous()
        self.reduction.weight = torch.nn.Parameter(self.reduction.weight.to(orig_device))
        if self.reduction.bias is not None:
            self.reduction.bias = torch.nn.Parameter(self.reduction.bias.to(orig_device))


        return x, (math.ceil(H / 2), math.ceil(W / 2))

def forward(self, x, mask=None):
    B, N, C = x.shape
    qkv = self.qkv(x).reshape(B, N, 3, self.num_heads, C // self.num_heads).permute(2, 0, 3, 1, 4)
    q, k, v = qkv[0], qkv[1], qkv[2]  # B_, nHead, N, C


    q = q * self.scale
    attn = (q @ k.transpose(-2, -1))


    relative_position_bias = self.relative_position_bias_table[self.relative_position_index.view(-1)].view(
        self.window_size[0] * self.window_size[1], self.window_size[0] * self.window_size[1], -1)
    relative_position_bias = relative_position_bias.permute(2, 0, 1).contiguous()  # nH, Wh*Ww, Wh*Ww
    attn = attn + relative_position_bias.unsqueeze(0)


    if mask is not None:
        nW = mask.shape[0]
        attn = attn.view(B // nW, nW, self.num_heads, N, N) + mask.unsqueeze(1).unsqueeze(0)
        attn = attn.view(-1, self.num_heads, N, N)
    attn = self.softmax(attn)


    attn = self.attn_drop(attn)
    x = (attn @ v).transpose(1, 2).reshape(B, N, C)
    x = self.proj(x)
    x = self.proj_drop(x)
    return x

通过「继承 + 动态替换」的方式，在你自己的训练代码（如tools/train.py）里重写PatchMerging的forward方法，然后动态替换模型中的 downsample 层 ——mmdet 原生库文件完全不动，别人使用时还是原生逻辑，只有你的训练会加载重写后的层。
具体步骤（无任何原生库修改，别人完全无感）

import math
import torch
import torch.nn as nn
import torch.nn.functional as F
# 导入原生PatchMerging（只导入，不修改）
from mmdet.models.utils.transformer import PatchMerging as OrigPatchMerging


# 继承原生类，重写forward方法（仅你的训练生效）
class A100PatchMerging(OrigPatchMerging):
    def forward(self, x, hw_shape):
        # 仅你的A100环境需要的CPU硬编码逻辑（原生库完全不变）
        orig_device = x.device
        x = x.cpu().float().contiguous()
        
        # reduction层权重临时转CPU
        self.reduction.weight = torch.nn.Parameter(self.reduction.weight.cpu().float())
        if self.reduction.bias is not None:
            self.reduction.bias = torch.nn.Parameter(self.reduction.bias.cpu().float())
        
        # 原生业务逻辑（完全复制，仅设备为CPU）
        H, W = hw_shape
        B, L, C = x.shape
        assert L == H * W, 'input feature has wrong size'


        x = x.view(B, H, W, C)
        pad_input = (H % 2 == 1) or (W % 2 == 1)
        if pad_input:
            x = F.pad(x, (0, 0, 0, W % 2, 0, H % 2))


        x0 = x[:, 0::2, 0::2, :]
        x1 = x[:, 1::2, 0::2, :]
        x2 = x[:, 0::2, 1::2, :]
        x3 = x[:, 1::2, 1::2, :]
        x = torch.cat([x0, x1, x2, x3], -1)
        x = x.view(B, -1, 4 * C)


        # CPU计算reduction
        x = self.reduction(x).float().contiguous()
        
        # 转回CUDA，恢复权重
        x = x.to(orig_device).float().contiguous()
        self.reduction.weight = torch.nn.Parameter(self.reduction.weight.to(orig_device))
        if self.reduction.bias is not None:
            self.reduction.bias = torch.nn.Parameter(self.reduction.bias.to(orig_device))


        return x, (math.ceil(H / 2), math.ceil(W / 2))


# ========== 新增：动态替换模型中的PatchMerging层 ==========
def replace_downsample_for_a100(model):
    for name, module in model.named_modules():
        # 找到所有downsample层（PatchMerging），替换为A100版本
        if isinstance(module, OrigPatchMerging):
            # 复制原生层的参数到新层
            a100_module = A100PatchMerging(
                in_channels=module.in_channels,
                out_channels=module.out_channels,
                norm_cfg=module.norm_cfg,
                init_cfg=module.init_cfg
            )
            a100_module.load_state_dict(module.state_dict())
            # 动态替换
            parent_name = name.rsplit('.', 1)[0] if '.' in name else ''
            parent_module = model.get_submodule(parent_name) if parent_name else model
            setattr(parent_module, name.split('.')[-1], a100_module)
    return model

def forward(self, x, input_size):
        """
        Args:
            x (Tensor): Has shape (B, H*W, C_in).
            input_size (tuple[int]): The spatial shape of x, arrange as (H, W).
                Default: None.


        Returns:
            tuple: Contains merged results and its spatial shape.


                - x (Tensor): Has shape (B, Merged_H * Merged_W, C_out)
                - out_size (tuple[int]): Spatial shape of x, arrange as
                    (Merged_H, Merged_W).
        """
 
        B, L, C = x.shape
        assert isinstance(input_size, Sequence), f'Expect ' \
                                                 f'input_size is ' \
                                                 f'`Sequence` ' \
                                                 f'but get {input_size}'


        H, W = input_size
        assert L == H * W, 'input feature has wrong size'


        x = x.view(B, H, W, C).permute([0, 3, 1, 2])  # B, C, H, W
        # Use nn.Unfold to merge patch. About 25% faster than original method,
        # but need to modify pretrained model for compatibility


        if self.adap_padding:
            x = self.adap_padding(x)
            H, W = x.shape[-2:]


        x = self.sampler(x)
        # if kernel_size=2 and stride=2, x should has shape (B, 4*C, H/2*W/2)


        out_h = (H + 2 * self.sampler.padding[0] - self.sampler.dilation[0] *
                 (self.sampler.kernel_size[0] - 1) -
                 1) // self.sampler.stride[0] + 1
        out_w = (W + 2 * self.sampler.padding[1] - self.sampler.dilation[1] *
                 (self.sampler.kernel_size[1] - 1) -
                 1) // self.sampler.stride[1] + 1


        output_size = (out_h, out_w)
        x = x.transpose(1, 2)  # B, H/2*W/2, 4*C
        x = self.norm(x) if self.norm else x
        x = self.reduction(x)
        return x, output_size

        input_size (tuple[int]): The spatial shape of x, arrange as (H, W).
            Default: None.


    Returns:
        tuple: Contains merged results and its spatial shape.


            - x (Tensor): Has shape (B, Merged_H * Merged_W, C_out)
            - out_size (tuple[int]): Spatial shape of x, arrange as
                (Merged_H, Merged_W).
    """
    # ========== A100兼容：强制转CPU+FP32（含norm层） ==========
    orig_device = x.device
    x = x.cpu().float().contiguous()
    
    # 1. reduction层权重转CPU
    self.reduction.weight = torch.nn.Parameter(self.reduction.weight.cpu().float())
    if self.reduction.bias is not None:
        self.reduction.bias = torch.nn.Parameter(self.reduction.bias.cpu().float())
    
    # 2. norm层权重转CPU（解决设备不匹配核心）
    if self.norm is not None:
        self.norm.weight = torch.nn.Parameter(self.norm.weight.cpu().float())
        self.norm.bias = torch.nn.Parameter(self.norm.bias.cpu().float())
    # ==========================================================


    B, L, C = x.shape
    assert isinstance(input_size, Sequence), f'Expect ' \
                                             f'input_size is ' \
                                             f'`Sequence` ' \
                                             f'but get {input_size}'


    H, W = input_size
    assert L == H * W, 'input feature has wrong size'


    x = x.view(B, H, W, C).permute([0, 3, 1, 2])  # B, C, H, W


    if self.adap_padding:
        x = self.adap_padding(x)
        H, W = x.shape[-2:]


    x = self.sampler(x)


    out_h = (H + 2 * self.sampler.padding[0] - self.sampler.dilation[0] *
             (self.sampler.kernel_size[0] - 1) -
             1) // self.sampler.stride[0] + 1
    out_w = (W + 2 * self.sampler.padding[1] - self.sampler.dilation[1] *
             (self.sampler.kernel_size[1] - 1) -
             1) // self.sampler.stride[1] + 1


    output_size = (out_h, out_w)
    x = x.transpose(1, 2)  # B, H/2*W/2, 4*C
    x = self.norm(x) if self.norm else x
    x = self.reduction(x)


    # ========== A100兼容：转回CUDA+恢复所有层权重 ==========
    x = x.to(orig_device).float().contiguous()
    
    # 1. 恢复reduction层到CUDA
    self.reduction.weight = torch.nn.Parameter(self.reduction.weight.to(orig_device))
    if self.reduction.bias is not None:
        self.reduction.bias = torch.nn.Parameter(self.reduction.bias.to(orig_device))
    
    # 2. 恢复norm层到CUDA
    if self.norm is not None:
        self.norm.weight = torch.nn.Parameter(self.norm.weight.to(orig_device))
        self.norm.bias = torch.nn.Parameter(self.norm.bias.to(orig_device))
    # ==========================================================


    return x, output_size
核心修改点：
新增了norm层的设备切换逻辑（共 4 行）：
计算前：把norm的weight/bias转 CPU+FP32，和输入x设备对齐；
计算后：把norm的参数转回 CUDA，和其他层保持一致；
这是解决LayerNorm设备不匹配的唯一关键，此前漏了这一步导致报错。

def forward(self, *args, **kwargs):
    """Forward function for LSS depth vtransform."""
    # ========== A100兼容：矩阵乘法转CPU ==========
    orig_device = args[0].device if args else next(self.parameters()).device
    
    # 提取并转CPU：lidar_aug_matrix
    if 'cur_lidar_aug_matrix' in kwargs:
        cur_lidar_aug_matrix = kwargs['cur_lidar_aug_matrix']
        mat_cpu = cur_lidar_aug_matrix[:3, :3].cpu().float()
        mat_inv_cpu = torch.inverse(mat_cpu)
    else:
        mat_inv_cpu = torch.eye(3).cpu().float()
